#!/usr/bin/env bash
set -euo pipefail

# Replayable offline packet build for product/test authority 51b97b8e.
# This script performs no provider call, production database operation,
# production App restart, merge, or push.

ROOT="$(git rev-parse --show-toplevel)"
PACKET="$ROOT/docs/superpowers/evidence/2026-08-27-lifecycle-honesty-repair"
PRODUCT_BASE="11e7a5d4f6856062a5ac00a8d90ed97b5c2e56cb"
PRODUCT_AUTHORITY="51b97b8e4f8d6f33aff495adc3423529b210a9d3"
APP_URL="http://127.0.0.1:4206/"
BASE_TREE="$(mktemp -d /tmp/arkscope-honesty-base.XXXXXX)"
SERVER_LOG="$(mktemp /tmp/arkscope-honesty-vite.XXXXXX.log)"
SERVER_PID=""

cleanup() {
  if [[ -n "$SERVER_PID" ]] && kill -0 "$SERVER_PID" 2>/dev/null; then
    kill "$SERVER_PID"
    wait "$SERVER_PID" || true
  fi
  rm -rf "$BASE_TREE"
  rm -f "$SERVER_LOG"
}
trap cleanup EXIT

cd "$ROOT"
rm -rf \
  "$PACKET/browser" \
  "$PACKET/SHA256SUMS" \
  "$PACKET/backend-focused-a.nodes" \
  "$PACKET/backend-focused-b.nodes" \
  "$PACKET/backend-focused-a.txt" \
  "$PACKET/backend-focused-b.txt" \
  "$PACKET/backend-full-a.txt" \
  "$PACKET/backend-full-b.txt" \
  "$PACKET/browser-matrix.json" \
  "$PACKET/frontend-build.txt" \
  "$PACKET/frontend-full.txt" \
  "$PACKET/frontend-i18n-literals.txt" \
  "$PACKET/frontend-typecheck.txt" \
  "$PACKET/mutation-ledger.json" \
  "$PACKET/offline-authority.json" \
  "$PACKET/schema-base.json" \
  "$PACKET/schema-comparison.json" \
  "$PACKET/schema-head.json" \
  "$PACKET/verification-summary.json"
find "$PACKET" -type d -name __pycache__ -prune -exec rm -rf -- {} +

# Apply and restore every mutation independently.
python "$PACKET/run_mutations.py" --output "$PACKET/mutation-ledger.json"

# Generate temporary-SQLite authority twice and compare exact bytes.
python "$PACKET/capture_offline_authority.py" \
  --output /tmp/arkscope-honesty-offline-authority-a.json
python "$PACKET/capture_offline_authority.py" \
  --output "$PACKET/offline-authority.json"
cmp /tmp/arkscope-honesty-offline-authority-a.json "$PACKET/offline-authority.json"
rm -f /tmp/arkscope-honesty-offline-authority-a.json

FOCUSED_TESTS=(
  tests/test_security_lifecycle_fact_kernel.py
  tests/test_security_lifecycle_sec_evidence.py
  tests/test_security_lifecycle_automation_scheduler.py
  tests/test_security_lifecycle_automation_worker.py
  tests/test_security_lifecycle_disposition.py
  tests/test_security_lifecycle_tools.py
  tests/test_security_lifecycle_routes.py
  tests/test_ticker_identity_transition.py
)

# Collect the focused node set twice, excluding only pytest's timing summary.
pytest --collect-only -q "${FOCUSED_TESTS[@]}" \
  | sed -E '/^[0-9]+ tests collected in [0-9.]+s$/d' \
  > "$PACKET/backend-focused-a.nodes"
pytest --collect-only -q "${FOCUSED_TESTS[@]}" \
  | sed -E '/^[0-9]+ tests collected in [0-9.]+s$/d' \
  > "$PACKET/backend-focused-b.nodes"
cmp "$PACKET/backend-focused-a.nodes" "$PACKET/backend-focused-b.nodes"

# Run focused and complete backend gates twice with isolated temp roots.
pytest --basetemp=/tmp/arkscope-honesty-focused-a "${FOCUSED_TESTS[@]}" -q \
  2>&1 | tee "$PACKET/backend-focused-a.txt"
pytest --basetemp=/tmp/arkscope-honesty-focused-b "${FOCUSED_TESTS[@]}" -q \
  2>&1 | tee "$PACKET/backend-focused-b.txt"
pytest --basetemp=/tmp/arkscope-honesty-full-a -q \
  2>&1 | tee "$PACKET/backend-full-a.txt"
pytest --basetemp=/tmp/arkscope-honesty-full-b -q \
  2>&1 | tee "$PACKET/backend-full-b.txt"

# Run complete frontend gates from the repository root.
npm --prefix apps/arkscope-web test \
  2>&1 | tee "$PACKET/frontend-full.txt"
npm --prefix apps/arkscope-web run typecheck \
  2>&1 | tee "$PACKET/frontend-typecheck.txt"
npm --prefix apps/arkscope-web run check:i18n-literals \
  2>&1 | tee "$PACKET/frontend-i18n-literals.txt"
npm --prefix apps/arkscope-web run build \
  2>&1 | tee "$PACKET/frontend-build.txt"

# Normalize generated text artifacts to one terminal newline.
for generated_text in \
  "$PACKET"/backend-focused-*.nodes \
  "$PACKET"/backend-focused-*.txt \
  "$PACKET"/backend-full-*.txt \
  "$PACKET"/frontend-*.txt; do
  perl -0pi -e 's/\n+\z/\n/' "$generated_text"
done

# Compare fresh in-memory schema authority with the local product-base archive.
git archive "$PRODUCT_BASE" src | tar -x -C "$BASE_TREE"
python "$PACKET/capture_profile_schema.py" \
  --repo "$BASE_TREE" \
  --revision "$PRODUCT_BASE" \
  --output "$PACKET/schema-base.json"
python "$PACKET/capture_profile_schema.py" \
  --repo "$ROOT" \
  --revision "$PRODUCT_AUTHORITY" \
  --output "$PACKET/schema-head.json"
python "$PACKET/compare_profile_schema.py" \
  --base "$PACKET/schema-base.json" \
  --head "$PACKET/schema-head.json" \
  --output "$PACKET/schema-comparison.json"

# Start only a transient loopback Vite fixture server and stop it after capture.
npm --prefix apps/arkscope-web run dev -- \
  --host 127.0.0.1 --port 4206 --strictPort >"$SERVER_LOG" 2>&1 &
SERVER_PID="$!"
for _ in $(seq 1 80); do
  if curl --fail --silent --show-error "$APP_URL" >/dev/null; then
    break
  fi
  sleep 0.25
done
kill -0 "$SERVER_PID"
ARKSCOPE_HONESTY_APP_URL="$APP_URL" python "$PACKET/run_browser_matrix.py"
kill "$SERVER_PID"
wait "$SERVER_PID" || true
SERVER_PID=""

# Summarize, seal the exact payload file set, and verify every digest.
python "$PACKET/write_verification_summary.py"
find "$PACKET" -type d -name __pycache__ -prune -exec rm -rf -- {} +
python "$PACKET/seal_packet.py"
(
  cd "$PACKET"
  sha256sum -c SHA256SUMS
)
